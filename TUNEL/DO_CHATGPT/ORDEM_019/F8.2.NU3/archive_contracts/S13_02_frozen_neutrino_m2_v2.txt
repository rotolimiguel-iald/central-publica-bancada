from __future__ import annotations

import hashlib
import json
import math
from datetime import date

# ----------------------------------------------------------------- constantes
ALPHA_FINE_CODATA_2018 = 7.2973525693e-3          # [INPUT] CODATA
ONE = 1.0                                          # omega(I) = 1 [POSTULATE]
SIN_45 = math.sin(math.radians(45.0))

# escala dimensional da identificacao — DECLARADA, nao derivada
MASS_SCALE_EV = 1.0                                # [NORM]

# ------------------------------------------------- medidas de Dm2_21, datadas
# (valor 1e-5 eV^2, sigma 1e-5 eV^2, rotulo, data, independente?)
DM2_21_MEASUREMENTS = [
    (7.53, 0.18, "PDG 2023 (a base do modulo v141)",          "2023",    False),
    (7.50, 0.12, "JUNO, primeiros resultados (59 dias)",       "2025-11", True),
    (7.53, 0.10, "NuFIT global pos-JUNO (dados ate fim 2025)", "2025-12", True),
]
DM2_21_PROJECTED = (7.53, 0.0377, "JUNO, meta de projeto 0,5%", "~2031", "[PROJECAO]")


def sha_obj(obj) -> str:
    return hashlib.sha256(
        json.dumps(obj, sort_keys=True, ensure_ascii=False, default=str).encode("utf-8")
    ).hexdigest()


def m2_from_dm2(dm2_1e5: float, sigma_1e5: float) -> tuple[float, float]:
    """m_2 = sqrt(Dm2_21) e sua incerteza propagada, em eV."""
    m = math.sqrt(dm2_1e5 * 1e-5)
    sm = (sigma_1e5 * 1e-5) / (2.0 * m)
    return m, sm


def prove_neutrino_m2_v2() -> dict:
    beta_tgl = ALPHA_FINE_CODATA_2018 * ONE * math.sqrt(math.e)   # jamais literal
    m2_pred = beta_tgl * SIN_45 * MASS_SCALE_EV                    # eV

    # ---- a tensao, medida contra cada determinacao datada -------------------
    tensions = []
    for v, s, rot, dt, indep in DM2_21_MEASUREMENTS:
        m_obs, s_obs = m2_from_dm2(v, s)
        tensions.append({
            "fonte": rot, "data": dt, "independente": indep,
            "Dm2_21_1e5": v, "sigma_1e5": s,
            "m2_obs_meV": m_obs * 1e3, "sigma_m2_meV": s_obs * 1e3,
            "tensao_sigma": abs(m_obs - m2_pred) / s_obs,
        })

    vp, sp, rotp, dtp, tagp = DM2_21_PROJECTED
    m_p, s_p = m2_from_dm2(vp, sp)
    projecao = {"fonte": rotp, "data": dtp, "estatuto": tagp,
                "sigma_m2_meV": s_p * 1e3,
                "tensao_projetada_sigma": abs(m_p - m2_pred) / s_p}

    corrente = tensions[-1]                     # a mais recente e independente

    # ---- a precisao que cruza a linha de morte ------------------------------
    delta = abs(m_p - m2_pred)                  # separacao em eV (valor central atual)
    sigma_kill = delta / 5.0                    # sigma_m2 que da 5 sigma
    dm2_sigma_kill = 2.0 * m_p * sigma_kill     # sigma em Dm2_21 (eV^2)
    prec_kill_pct = 100.0 * dm2_sigma_kill / (vp * 1e-5)

    # ---- por que Sigma m_nu foi APOSENTADO como discriminante ---------------
    d32, d32e = 2.437e-3, 0.028e-3               # PDG 2023 [EXT]
    m3_tgl = math.sqrt(m2_pred ** 2 + d32)
    sum_tgl = m2_pred + m3_tgl
    sum_err = 0.5 * d32e / m3_tgl
    m2_std = math.sqrt(7.53e-5)
    m3_std = math.sqrt(7.53e-5 + d32)
    sum_std = m2_std + m3_std
    sep_sigma = abs(sum_tgl - sum_std) / sum_err

    aposentadoria = {
        "observavel_aposentado": "Sigma m_nu (frozen NEUTRINO_SUM_V1, um.py:3929, v147)",
        "Sigma_TGL_meV": sum_tgl * 1e3,
        "Sigma_NH_padrao_m1zero_meV": sum_std * 1e3,
        "separacao_meV": abs(sum_tgl - sum_std) * 1e3,
        "sigma_da_predicao_meV": sum_err * 1e3,
        "separacao_em_sigmas_da_predicao": sep_sigma,
        "motivo": ("a predicao TGL em Sigma dista %.3f meV do ordenamento normal minimo "
                   "(m_1=0), o que e %.2f sigma da propria barra da predicao. Todo limite "
                   "que cruza a linha de morte do V1 tambem exclui o NH padrao. Portanto "
                   "Sigma m_nu nao distingue a TGL do minimo permitido pela oscilacao."
                   % (abs(sum_tgl - sum_std) * 1e3, sep_sigma)),
        "estatuto": "NAO_DISCRIMINANTE — aposentado como teste de beta_TGL",
    }

    # ---- o frozen ----------------------------------------------------------
    frozen = {
        "version": "NEUTRINO_M2_V2",
        "supersedes": "NEUTRINO_SUM_V1 (um.py:3929, v147) como TESTE DE beta_TGL",
        "data_do_congelamento": str(date.today()),
        "observavel": "m_2 = sqrt(Dm2_21), a massa do 2o autoestado no ordenamento normal",
        "predicao": ("m_2 = beta_TGL * sin(45 graus) * 1 eV, com beta_TGL = alpha_fine*sqrt(e) "
                     "em runtime [derivacao publicada 2025, Zenodo 10.5281/zenodo.17526619]"),
        "escala_dimensional": ("1 eV — [NORM] DECLARADA, NAO DERIVADA. A escala que zeraria a "
                               "tensao contra o NuFIT pos-JUNO seria 1,0200 eV (+2,00%). "
                               "Esta e a unica liberdade da formula e ela esta fixada por "
                               "declaracao, nao por ajuste."),
        "kill_rule": ("FALSIFIED sse |m_2_obs - m_2_TGL| / sigma_obs >= 5 em DUAS determinacoes "
                      "independentes de Dm2_21 (robustez de experimento, nao so de modelo)"),
        "tension_rule": ("TENSION sse a mesma razao ficar em [3, 5) na determinacao global "
                        "mais recente"),
        "matched_rule": ("MATCHED_PROSPECTIVELY sse duas determinacoes independentes pousarem "
                         "dentro de 1 sigma de m_2_TGL — e mesmo entao NAO e CONFIRMED"),
        "allowed_verdicts": ["TGL_NU_M2_ARMED_CONSISTENT",
                             "TGL_NU_M2_TENSION",
                             "TGL_NU_M2_FALSIFIED",
                             "TGL_NU_M2_MATCHED_PROSPECTIVELY"],
        "forbidden_verdicts": ["CONFIRMED"],
        "custody": ("(i) a derivacao de m_2 antecede o dado de JUNO (Zenodo 2025 vs JUNO nov/2025); "
                    "(ii) a reespecificacao de Sigma para m_2 foi decidida por NAO-DISCRIMINACAO, "
                    "medida ANTES de consultar limites novos de Sigma; "
                    "(iii) a escala de 1 eV e anterior a esta rodada e nao foi tocada."),
        "NAO_E_CEGO_NESTA_EMISSAO": (
            "DECLARACAO OBRIGATORIA PELA REGUA DA CASA. Os limiares desta regra (5 sigma para "
            "FALSIFIED, 3 sigma para TENSION) foram fixados na MESMA sessao em que a tensao "
            "corrente (2,95 sigma) ja havia sido calculada. Portanto a emissao de HOJE nao e um "
            "veredito cego: e uma REESPECIFICACAO. A cegueira vale a partir de agora — as "
            "proximas determinacoes de Dm2_21 serao julgadas por uma regra ja hasheada. "
            "Registrar isto e o que impede a reespecificacao de virar ajuste."),
        "does_not_gate_core": True,
    }
    frozen_hash = sha_obj(frozen)

    # ---- veredito ----------------------------------------------------------
    t_now = corrente["tensao_sigma"]
    n_indep_5sigma = sum(1 for t in tensions if t["independente"] and t["tensao_sigma"] >= 5.0)
    if n_indep_5sigma >= 2:
        verdict = "TGL_NU_M2_FALSIFIED"
    elif t_now >= 3.0:
        verdict = "TGL_NU_M2_TENSION"
    else:
        verdict = "TGL_NU_M2_ARMED_CONSISTENT"

    checks = [
        ("beta_TGL calculado em runtime, jamais literal: %.10f" % beta_tgl, True),
        ("m_2 predito = %.4f meV (escala 1 eV [NORM])" % (m2_pred * 1e3), True),
        ("Sigma m_nu APOSENTADO como discriminante: separacao %.3f meV = %.2f sigma"
         % (abs(sum_tgl - sum_std) * 1e3, sep_sigma), sep_sigma < 1.0),
        ("tensao corrente (%s): %.2f sigma" % (corrente["fonte"], t_now), True),
        ("a tensao CRESCEU com a precisao: %.2f -> %.2f -> %.2f sigma"
         % (tensions[0]["tensao_sigma"], tensions[1]["tensao_sigma"], tensions[2]["tensao_sigma"]), True),
        ("linha de morte alcancada quando Dm2_21 for medido a %.2f%% (JUNO projeta 0,5%%)"
         % prec_kill_pct, True),
        ("CONFIRMED proibido; o gate NAO se move por este canal", True),
    ]
    all_v = bool(all(v for _, v in checks))

    return {
        "theorem": ("O TESTE DO NEUTRINO REESPECIFICADO: o discriminante e m_2, nao Sigma. "
                    "m_2(TGL) = %.4f meV contra %.4f +- %.4f meV (%s) => %.2f sigma. "
                    "A tensao cresce com a precisao e a linha de morte de 5 sigma e alcancada "
                    "quando Dm2_21 chegar a %.2f%% — dentro da meta de projeto do JUNO."
                    % (m2_pred * 1e3, corrente["m2_obs_meV"], corrente["sigma_m2_meV"],
                       corrente["fonte"], t_now, prec_kill_pct)),
        "frozen": frozen,
        "frozen_hash": frozen_hash,
        "values": {
            "beta_TGL": beta_tgl,
            "m2_pred_eV": m2_pred,
            "m2_pred_meV": m2_pred * 1e3,
            "escala_que_zeraria_a_tensao_eV": (m_p / SIN_45) / beta_tgl,
            "precisao_Dm2_21_que_mata_pct": prec_kill_pct,
            "sigma_m2_que_mata_meV": sigma_kill * 1e3,
        },
        "tensoes_datadas": tensions,
        "projecao": projecao,
        "aposentadoria_do_Sigma": aposentadoria,
        "checks": checks,
        "all_verified": all_v,
        "does_not_gate_core": True,
        "verdict": verdict,
    }
